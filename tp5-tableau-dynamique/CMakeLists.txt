cmake_minimum_required(VERSION 3.20)
project(tp5_tableau_dynamique CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(tp5 src/main.cpp)
option(USE_SANITIZERS "Activer ASan + UBSan" OFF)
if(USE_SANITIZERS)
  target_compile_options(tp5 PRIVATE -fsanitize=address,undefined -g)
  target_link_options(tp5 PRIVATE -fsanitize=address,undefined)
endif()
